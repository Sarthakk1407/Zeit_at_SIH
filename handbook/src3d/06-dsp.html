boom[i] : 0; if (!impulseNear && sIn0 > dbToPa(70) ** 2 * BLK) accSnr(sIn0, nIn0, sIn0, nIn0); dsp.prevBoom.set(boom); dsp.prevClean.set(clean); dsp.prevNoise.set(noise); return frame; } // B4 STFT of the mixture, and of the clean and noise parts separately so the // SNR figures are measured, not guessed loadFrame(dsp.re, dsp.im, dsp.prevBoom, boom); loadFrame(dsp.re2, dsp.im2, dsp.prevClean, clean); loadFrame(dsp.re3, dsp.im3, dsp.prevNoise, noise); fft(dsp.re, dsp.im, false); fft(dsp.re2, dsp.im2, false); fft(dsp.re3, dsp.im3, false); // throat envelope — the voice guard's evidence (B9) let thr = 0; for (let i = 0; i < BLK; i++) thr += throat[i] * throat[i]; dsp.throatEnv = dsp.throatEnv * .6 + Math.sqrt(thr / BLK) * .4; const voiced = paToDb(dsp.throatEnv) > 62; m.voiced = voiced ? 1 : 0; let sOut = 0, nOut = 0, gSum = 0; for (let k = 0; k <= NFFT / 2; k++) { const Nraw = dsp.re3[k] * dsp.re3[k] + dsp.im3[k] * dsp.im3[k]; const P = (dsp.re[k] * dsp.re[k] + dsp.im[k] * dsp.im[k]) * inv2; // B3: normalised power const N = Nraw * inv2; dsp.noisePsd[k] = dsp.noisePsd[k] * .92 + (voiced ? dsp.noisePsd[k] * .08 : N * .08); let g = 1 - 1.6 * dsp.noisePsd[k] / (P + 1e-18); const fHz = k * SR / NFFT; let floorG = impulse ? .25 : .06; // B6/B8: repair rather than gouge // B9 voice guard — not on a frame with a gunshot in it: a 145 dB crack reaches even a // contact microphone, and the guard would then protect the crack instead of the voice if (voiced && !impulse && fHz > 250 && fHz < 3600) floorG = Math.max(floorG, .35); g = Math.max(floorG, Math.min(1, g)); g = dsp.gPrev[k] * .45 + g * .55; dsp.gPrev[k] = g; gSum += g; const gg = g * g; sOut += (dsp.re2[k] * dsp.re2[k] + dsp.im2[k] * dsp.im2[k]) * gg; nOut += Nraw * gg; dsp.re[k] *= g; dsp.im[k] *= g; if (k > 0 && k < NFFT / 2) { dsp.re[NFFT - k] = dsp.re[k]; dsp.im[NFFT - k] = -dsp.im[k]; } } fft(dsp.re, dsp.im, true); // B11 overlap-add, then the PTT gate — nothing leaves unless the button is down for (let i = 0; i < BLK; i++) { const v = dsp.ola[i] + dsp.re[i] * WIN[i]; txOut[i] = s.ptt ? v : 0; dsp.ola[i] = dsp.re[i + BLK] * WIN[i + BLK]; } // SNR is a speech metric: only where the wearer is speaking and no shot is in the frame if (!impulseNear && sIn0 > dbToPa(70) ** 2 * BLK) accSnr(sIn0, nIn0, sOut, nOut); if (impulse && !dsp.wasImpulse) m.shots++; dsp.wasImpulse = impulse; dsp.prevBoom.set(boom); dsp.prevClean.set(clean); dsp.prevNoise.set(noise); frame.sOut = sOut; frame.nOut = nOut; frame.voiced = voiced; frame.maskMean = gSum / NBIN; return frame; } /* SNR over about two seconds of speech and noise energy; the output figures come from pushing the clean part and the noise part through the same mask */ function accSnr(sIn, nIn, sOut, nOut) { const b = 0.008; dsp.sInAcc += (sIn - dsp.sInAcc) * b; dsp.nInAcc += (nIn - dsp.nInAcc) * b; dsp.sOutAcc += (sOut - dsp.sOutAcc) * b; dsp.nOutAcc += (nOut - dsp.nOutAcc) * b; dsp.silent = sIn < dbToPa(55) ** 2 * BLK ? dsp.silent + 1 : 0; const quiet = dsp.silent > 45; S.m.snrIn = quiet ? NaN : 10 * Math.log10((dsp.sInAcc + 1e-20) / (dsp.nInAcc + 1e-20)); S.m.snrOut = quiet ? NaN : 10 * Math.log10((dsp.sOutAcc + 1e-20) / (dsp.nOutAcc + 1e-20)); } S.set = function (patch) { const before = JSON.stringify(S.s); Object.assign(S.s, patch); if (patch.scene && JSON.stringify(S.s) !== before) logLine('e', `scene → ${SCENE[S.s.scene].label}`); if ('anc' in patch) logLine('e', `Lane A ANC ${patch.anc ? 'on' : 'off'}`); if ('clean' in patch) logLine('e', `Lane B clean-up ${patch.clean ? 'on' : 'off'}`); if ('adapt' in patch && !patch.adapt) logLine('e', 'adaptation frozen by hand'); if (opt.live && typeof syncControls === 'function') syncControls(); }; S.reset = function () { dsp.W.fill(0); logLine('e', 'canceller reset'); }; return S; } function loadFrame(re, im, prev, cur) { for (let i = 0; i < BLK; i++) { re[i] = prev[i] * WIN[i]; im[i] = 0; } for (let i = 0; i < BLK; i++) { re[i + BLK] = cur[i] * WIN[i + BLK]; im[i + BLK] = 0; } } const SIM = createMachine({ live: true }); SIM.logLine('e', 'bench powered — STM32 loads the DSP program into the ADAU over I²C');
